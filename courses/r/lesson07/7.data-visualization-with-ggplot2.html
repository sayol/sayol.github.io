ggplot2::ggplot(
  data = stats::na.omit(penguins),
  mapping = ggplot2::aes(x = flipper_len, y = body_mass)
) +
  ggplot2::labs(
    title = "Relationship of Flipper Length and Body Mass",
    x = "Flipper Length (mm)",
    y = "Body Mass (g)",
  ) +
  ggplot2::geom_point(
    mapping = ggplot2::aes(fill = sex),
    shape = 21,
    size = 4,
    alpha = 0.75
  ) +
  ggplot2::geom_smooth(method = "lm", formula = y ~ x) +
  ggplot2::facet_grid(
    rows = ggplot2::vars(species),
    cols = ggplot2::vars(sex)
  ) +
  # facet_grid(species~sex) also work
  ggplot2::scale_fill_manual(values = c("red", "blue")) +
  ggplot2::theme_light() +
  ggplot2::theme(
    plot.title = ggplot2::element_text(
      family = "Arial",
      size = 20,
      face = "bold"
    ),
    axis.title = ggplot2::element_text(
      family = "Consolas",
      size = 18,
      face = "bold"
    ),
    axis.text = ggplot2::element_text(
      family = "Consolas",
      size = 14,
      color = "black"
    ),
    strip.text = ggplot2::element_text(
      family = "Arial",
      size = 16,
      color = "brown"
    ),
    legend.title = ggplot2::element_text(family = "Arial", size = 20),
    legend.text = ggplot2::element_text(family = "Consolas", size = 14)
  )

penguins$year <- base::as.factor(x = penguins$year)
penguins %>%
  base::sapply(X = ., FUN = is.numeric) %>%
  penguins[, .] %>%
  stats::na.omit(object = .) %>%
  stats::cor(x = .) %>%
  base::as.data.frame(x = .) %>%
  dplyr::mutate(., "row" = base::rownames(.)) %>%
  tidyr::pivot_longer(
    data = .,
    cols = -"row",
    names_to = "col",
    values_to = "cor"
  ) %>%
  ggplot2::ggplot(data = ., ggplot2::aes(x = row, y = col, fill = cor)) +
  ggplot2::geom_tile() +
  ggplot2::geom_text(
    mapping = ggplot2::aes(label = base::sprintf("%.2f", cor)),
    color = "black",
    size = 5
  ) +
  ggplot2::scale_fill_gradient2(
    low = "blue", high = "red", mid = "white",
    midpoint = 0, limit = base::c(-1, 1), space = "Lab",
    name = "Pearson\nCorrelation"
  ) +
  ggplot2::theme_minimal() +
  ggplot2::labs(x = "", y = "") +
  ggplot2::theme(
    axis.text.x = ggplot2::element_text(angle = 45, vjust = 1, hjust = 1)
  )
